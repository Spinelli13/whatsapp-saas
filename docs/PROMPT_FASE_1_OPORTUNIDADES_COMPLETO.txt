const { Oportunidade, EstagioPipeline, HistoricoOportunidade, Usuario } = require('../models')
const { Op } = require('sequelize')

class OportunidadeService {
  async criar(clienteId, dados, usuarioId) {
    const oportunidade = await Oportunidade.create({
      cliente_id: clienteId,
      usuario_id: usuarioId,
      ...dados
    })

    await this.registrarHistorico(oportunidade.id, usuarioId, 'Oportunidade criada')
    return oportunidade
  }

  async listar(clienteId, filtros = {}) {
    const where = { cliente_id: clienteId }

    if (filtros.status) where.status = filtros.status
    if (filtros.usuario_id) where.usuario_id = filtros.usuario_id
    if (filtros.estagio_id) where.estagio_id = filtros.estagio_id

    return await Oportunidade.findAll({
      where,
      include: [
        { model: EstagioPipeline, as: 'estagio' },
        { model: Usuario, as: 'usuario', attributes: ['id', 'nome', 'email'] }
      ],
      order: [['posicao_coluna', 'ASC'], ['criado_em', 'DESC']],
      limit: filtros.limit || 100
    })
  }

  async obter(oportunidadeId, clienteId) {
    return await Oportunidade.findOne({
      where: {
        id: oportunidadeId,
        cliente_id: clienteId
      },
      include: [
        { model: EstagioPipeline, as: 'estagio' },
        { model: Usuario, as: 'usuario', attributes: ['id', 'nome', 'email'] },
        { 
          model: HistoricoOportunidade, 
          as: 'historico',
          include: [{ model: Usuario, as: 'usuario' }]
        }
      ]
    })
  }

  async atualizar(oportunidadeId, clienteId, dados, usuarioId) {
    const oportunidade = await this.obter(oportunidadeId, clienteId)
    if (!oportunidade) throw new Error('Oportunidade não encontrada')

    for (const [campo, valor] of Object.entries(dados)) {
      if (oportunidade[campo] !== valor) {
        await this.registrarHistorico(
          oportunidadeId,
          usuarioId,
          `${campo} alterado`,
          oportunidade[campo],
          valor
        )
      }
    }

    await oportunidade.update(dados)
    return oportunidade
  }

  async moverParaEstagio(oportunidadeId, clienteId, novoEstagioId, usuarioId, novaPosicao) {
    const oportunidade = await this.obter(oportunidadeId, clienteId)
    if (!oportunidade) throw new Error('Oportunidade não encontrada')

    const estagio = await EstagioPipeline.findOne({
      where: { id: novoEstagioId, cliente_id: clienteId }
    })
    if (!estagio) throw new Error('Estágio não encontrado')

    const estadoAnterior = oportunidade.estagio_id
    await oportunidade.update({
      estagio_id: novoEstagioId,
      posicao_coluna: novaPosicao || 0
    })

    await this.registrarHistorico(
      oportunidadeId,
      usuarioId,
      `Movido de ${estadoAnterior} para ${novoEstagioId}`
    )

    return oportunidade
  }

  async marcarGanha(oportunidadeId, clienteId, usuarioId) {
    return await this.atualizar(
      oportunidadeId,
      clienteId,
      {
        status: 'ganha',
        data_fechamento_real: new Date()
      },
      usuarioId
    )
  }

  async marcarPerdida(oportunidadeId, clienteId, motivoPerda, usuarioId) {
    return await this.atualizar(
      oportunidadeId,
      clienteId,
      {
        status: 'perdida',
        motivo_perda: motivoPerda
      },
      usuarioId
    )
  }

  async obterEstatisticas(clienteId) {
    const total = await Oportunidade.count({
      where: { cliente_id: clienteId }
    })

    const ganhas = await Oportunidade.count({
      where: { cliente_id: clienteId, status: 'ganha' }
    })

    const valorEsperado = await Oportunidade.findAll({
      where: {
        cliente_id: clienteId,
        status: 'aberta'
      },
      raw: true
    }).then(opor => 
      opor.reduce((acc, o) => acc + (parseFloat(o.valor || 0) * (o.probabilidade / 100)), 0)
    )

    return {
      total,
      ganhas,
      taxaGanho: total > 0 ? ((ganhas / total) * 100).toFixed(1) : 0,
      valorEsperado: valorEsperado.toFixed(2)
    }
  }

  async registrarHistorico(oportunidadeId, usuarioId, acao, valorAnterior = null, valorNovo = null) {
    await HistoricoOportunidade.create({
      oportunidade_id: oportunidadeId,
      usuario_id: usuarioId,
      acao,
      valor_anterior: valorAnterior,
      valor_novo: valorNovo
    })
  }

  async deletar(oportunidadeId, clienteId) {
    return await Oportunidade.destroy({
      where: {
        id: oportunidadeId,
        cliente_id: clienteId
      }
    })
  }
}

module.exports = new OportunidadeService()

const { EstagioPipeline, Oportunidade } = require('../models')

class PipelineService {
  async listarEstagios(clienteId) {
    return await EstagioPipeline.findAll({
      where: { cliente_id: clienteId, ativo: true },
      order: [['ordem', 'ASC']]
    })
  }

  async criarEstagio(clienteId, dados) {
    const proximaOrdem = await EstagioPipeline.max('ordem', {
      where: { cliente_id: clienteId }
    }) || 0

    return await EstagioPipeline.create({
      cliente_id: clienteId,
      ordem: proximaOrdem + 1,
      ...dados
    })
  }

  async atualizarEstagio(estagioId, clienteId, dados) {
    await EstagioPipeline.update(dados, {
      where: { id: estagioId, cliente_id: clienteId }
    })
    return this.obterEstagio(estagioId, clienteId)
  }

  async obterEstagio(estagioId, clienteId) {
    return await EstagioPipeline.findOne({
      where: { id: estagioId, cliente_id: clienteId }
    })
  }

  async deletarEstagio(estagioId, clienteId) {
    await EstagioPipeline.destroy({
      where: { id: estagioId, cliente_id: clienteId }
    })
  }

  async obterPipelineCompleto(clienteId) {
    const estagios = await this.listarEstagios(clienteId)
    
    const pipeline = await Promise.all(
      estagios.map(async (estagio) => {
        const oportunidades = await Oportunidade.findAll({
          where: {
            cliente_id: clienteId,
            estagio_id: estagio.id
          },
          order: [['posicao_coluna', 'ASC']],
          attributes: ['id', 'titulo', 'valor', 'probabilidade', 'data_fechamento_esperada']
        })

        return {
          ...estagio.toJSON(),
          oportunidades,
          total: oportunidades.length,
          valor_total: oportunidades.reduce((acc, o) => acc + parseFloat(o.valor || 0), 0)
        }
      })
    )

    return pipeline
  }
}

module.exports = new PipelineService()

  const carregarDados = async () => {
    try {
      setLoading(true)
      const [pipelineRes, statsRes] = await Promise.all([
        fetch('/api/oportunidades/pipeline').then(r => r.json()),
        fetch('/api/oportunidades/stats/dashboard').then(r => r.json())
      ])
      setPipeline(pipelineRes)
      setStats(statsRes)
    } catch (error) {
      console.error('Erro:', error)
    } finally {
      setLoading(false)
    }
  }

  return (
    <div className={`p-6 space-y-6 ${theme === 'dark' ? 'bg-slate-950' : 'bg-gray-50'}`}>
      <div className="flex items-center justify-between">
        <h2 className={`text-2xl font-bold ${theme === 'dark' ? 'text-white' : 'text-gray-900'}`}>
          Oportunidades
        </h2>
        <button className="flex items-center gap-2 px-4 py-2 rounded-lg bg-gradient-to-r from-cyan-600 to-blue-600 text-white hover:shadow-lg transition-all">
          <Plus className="w-4 h-4" /> Nova Oportunidade
        </button>
      </div>

      <div className="flex gap-2 border-b">
        <button
          onClick={() => setView('kanban')}
          className={`px-4 py-2 border-b-2 transition-colors ${
            view === 'kanban'
              ? `border-cyan-500 ${theme === 'dark' ? 'text-cyan-400' : 'text-cyan-600'}`
              : `border-transparent ${theme === 'dark' ? 'text-slate-400' : 'text-gray-600'}`
          }`}
        >
          Pipeline Kanban
        </button>
        <button
          onClick={() => setView('stats')}
          className={`px-4 py-2 border-b-2 transition-colors ${
            view === 'stats'
              ? `border-cyan-500 ${theme === 'dark' ? 'text-cyan-400' : 'text-cyan-600'}`
              : `border-transparent ${theme === 'dark' ? 'text-slate-400' : 'text-gray-600'}`
          }`}
        >
          Estatísticas
        </button>
      </div>

      {stats && (
        <div className="grid grid-cols-1 md:grid-cols-4 gap-4">
          <StatsCard label="Total" value={stats.total} icon={BarChart3} theme={theme} />
          <StatsCard label="Ganhas" value={stats.ganhas} icon={CheckCircle} theme={theme} />
          <StatsCard label="Taxa Ganho" value={`${stats.taxaGanho}%`} icon={TrendingUp} theme={theme} />
          <StatsCard label="Valor Esperado" value={`R$ ${parseFloat(stats.valorEsperado).toLocaleString()}`} icon={DollarSign} theme={theme} />
        </div>
      )}

      {view === 'kanban' && !loading && <PipelineBoard pipeline={pipeline} onRefresh={carregarDados} />}
    </div>
  )
}

const StatsCard: React.FC<any> = ({ label, value, icon: Icon, theme }) => (

    <div className="flex items-start justify-between">

        <p className={`text-sm ${theme === 'dark' ? 'text-slate-400' : 'text-gray-600'}`}>{label}</p>
        <p className={`text-2xl font-bold mt-2 ${theme === 'dark' ? 'text-white' : 'text-gray-900'}`}>{value}</p>
      </div>
      {Icon && <Icon className="w-6 h-6 text-cyan-400" />}
    </div>
  </div>

import React, { useState } from 'react'
import { useThemeStore } from '../../store/themeStore'
import { OportunidadeCard } from './OportunidadeCard'

export const PipelineBoard: React.FC<{ pipeline: any[], onRefresh: () => void }> = ({ pipeline, onRefresh }) => {
  const { theme } = useThemeStore()
  const [draggedItem, setDraggedItem] = useState(null)

  const handleDragStart = (e: any, oportunidade: any) => {
    setDraggedItem(oportunidade)
    e.dataTransfer.effectAllowed = 'move'
  }

  const handleDragOver = (e: any) => {
    e.preventDefault()
    e.dataTransfer.dropEffect = 'move'
  }

  const handleDrop = async (e: any, estagioId: string) => {
    e.preventDefault()
    if (!draggedItem) return

    try {
      await fetch(`/api/oportunidades/${draggedItem.id}/mover`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ estagio_id: estagioId, posicao: 0 })
      })
      onRefresh()
    } catch (error) {
      console.error('Erro:', error)
    }
  }

  return (
    <div className={`overflow-x-auto pb-4`} style={{ display: 'flex', gap: '1rem' }}>
      {pipeline.map((estagio) => (
        <div
          key={estagio.id}
          className={`
            p-4 rounded-lg border-2 border-dashed flex-shrink-0 min-h-96
            ${theme === 'dark'
              ? 'bg-slate-900 border-slate-700 hover:border-slate-600'
              : 'bg-white border-gray-300 hover:border-gray-400'
            }
          `}
          style={{ width: '350px' }}
          onDragOver={handleDragOver}
          onDrop={(e) => handleDrop(e, estagio.id)}
        >
          <div className="mb-4">
            <div className="flex items-center gap-2">
              <div className="w-3 h-3 rounded-full" style={{ backgroundColor: estagio.cor }} />
              <h3 className={`font-semibold ${theme === 'dark' ? 'text-white' : 'text-gray-900'}`}>
                {estagio.nome}
              </h3>
            </div>
            <p className={`text-xs mt-1 ${theme === 'dark' ? 'text-slate-500' : 'text-gray-500'}`}>
              {estagio.total} • R$ {parseFloat(estagio.valor_total).toLocaleString()}
            </p>
          </div>

          <div className="space-y-3">
            {estagio.oportunidades.map((op: any) => (
              <OportunidadeCard
                key={op.id}
                oportunidade={op}
                theme={theme}
                onDragStart={(e: any) => handleDragStart(e, op)}
              />
            ))}
          </div>
        </div>
      ))}
    </div>
  )
}
